### JAVASCRIPT - CONCEPTOS ESENCIALES

#### 1. Closures
```javascript
// Pregunta típica: ¿Qué imprime este código?
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 1000);
}
// Respuesta: 3, 3, 3 (porque var no tiene block scope)

// Solución con closure:
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 1000);
}
// Respuesta: 0, 1, 2

// Explicación: Closure es cuando una función tiene acceso a variables
// de su scope externo incluso después de que la función externa haya terminado
```

async function fetchMultipleUsers(userIds) {
  try {
    const promises = userIds.map(id => fetch(`/api/users/${id}`));
    const responses = await Promise.all(promises);
    const users = await Promise.all(responses.map(r => r.json()));
    return users;

  }
}

// Pregunta: ¿Cuál es la diferencia entre Promise.all y Promise.allSettled?

```javascript

// Respuesta: 1, 4, 3, 2

#### 4. Array Methods

// Pregunta: Implementa map, filter, reduce desde cero

Array.prototype.myMap = function(callback) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    result.push(callback(this[i], i, this));
  }
  return result;
};

Array.prototype.myFilter = function(callback) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (callback(this[i], i, this)) {
      result.push(this[i]);
    }
  }
  return result;
};

Array.prototype.myReduce = function(callback, initialValue) {
  let accumulator = initialValue !== undefined ? initialValue : this[0];
  let startIndex = initialValue !== undefined ? 0 : 1;
  
  for (let i = startIndex; i < this.length; i++) {
    accumulator = callback(accumulator, this[i], i, this);
  }
  return accumulator;
};
```

### REACT - CONCEPTOS ESENCIALES

#### 1. useState y useEffect
```javascript
// Pregunta: ¿Qué problema tiene este código?
function BadComponent() {
  const [count, setCount] = useState(0);
  
  useEffect(() => {
    setCount(count + 1); // ❌ Infinite loop!
  });
  
  return <div>{count}</div>;
}

// Solución:
function GoodComponent() {
  const [count, setCount] = useState(0);
  
  useEffect(() => {
    setCount(c => c + 1); // ✅ Solo se ejecuta una vez
  }, []); // ✅ Dependency array vacío
  
  return <div>{count}</div>;
}
```

#### 2. Custom Hooks
```javascript
// Pregunta: Crea un custom hook para fetch data

function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  
  useEffect(() => {
    const fetchData = async () => {
      try {
        setLoading(true);
        const response = await fetch(url);
        if (!response.ok) throw new Error('Network response was not ok');
        const json = await response.json();
        setData(json);
        setError(null);
      } catch (err) {
        setError(err.message);
        setData(null);
      } finally {
        setLoading(false);
      }
    };
    
    fetchData();
  }, [url]);
  
  return { data, loading, error };
}

// Uso:
function UserProfile({ userId }) {
  const { data, loading, error } = useFetch(`/api/users/${userId}`);
  
  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error: {error}</div>;
  return <div>{data.name}</div>;
}
```

#### 3. Performance Optimization
```javascript
// Pregunta: Optimiza este componente

// ❌ Problema: Re-renderiza innecesariamente
function ParentComponent() {
  const [count, setCount] = useState(0);
  
  const expensiveCalculation = () => {
    console.log('Calculating...');
    return count * 2;
  };
  
  return (
    <div>
      <button onClick={() => setCount(count + 1)}>Increment</button>
      <ChildComponent value={expensiveCalculation()} />
    </div>
  );
}

// ✅ Solución: useMemo y React.memo
function ParentComponent() {
  const [count, setCount] = useState(0);
  
  const expensiveValue = useMemo(() => {
    console.log('Calculating...');
    return count * 2;
  }, [count]);
  
  return (
    <div>
      <button onClick={() => setCount(count + 1)}>Increment</button>
      <ChildComponent value={expensiveValue} />
    </div>
  );
}

const ChildComponent = React.memo(({ value }) => {
  console.log('Child rendering');
  return <div>{value}</div>;
});
```

### ALGORITMOS - PROBLEMAS COMUNES

#### 1. Two Sum (LeetCode Easy)
```javascript
// Pregunta: Encuentra dos números que sumen target
// Input: nums = [2,7,11,15], target = 9
// Output: [0,1] (porque nums[0] + nums[1] = 9)

// Solución O(n) con hash map:
function twoSum(nums, target) {
  const map = new Map();
  
  for (let i = 0; i < nums.length; i++) {
    const complement = target - nums[i];
    
    if (map.has(complement)) {
      return [map.get(complement), i];
    }
    
    map.set(nums[i], i);
  }
  
  return [];
}

// Time: O(n), Space: O(n)
```

#### 2. Reverse String
```javascript
// Pregunta: Invierte un string

// Solución 1: Built-in methods
function reverseString1(s) {
  return s.split('').reverse().join('');
}

// Solución 2: Two pointers (in-place si es array)
function reverseString2(s) {
  const arr = s.split('');
  let left = 0;
  let right = arr.length - 1;
  
  while (left < right) {
    [arr[left], arr[right]] = [arr[right], arr[left]];
    left++;
    right--;
  }
  
  return arr.join('');
}

// Solución 3: Recursion
function reverseString3(s) {
  if (s.length <= 1) return s;
  return reverseString3(s.slice(1)) + s[0];
}
```

#### 3. Palindrome
```javascript
// Pregunta: Verifica si un string es palíndromo

function isPalindrome(s) {
  // Limpiar: solo alfanuméricos, lowercase
  const cleaned = s.toLowerCase().replace(/[^a-z0-9]/g, '');
  
  let left = 0;
  let right = cleaned.length - 1;
  
  while (left < right) {
    if (cleaned[left] !== cleaned[right]) {
      return false;
    }
    left++;
    right--;
  }
  
  return true;
}

// Test:
console.log(isPalindrome("A man, a plan, a canal: Panama")); // true
```

#### 5. Fibonacci
```javascript
// Pregunta: Calcula el n-ésimo número de Fibonacci

// Solución 1: Recursion (ineficiente O(2^n))
function fib1(n) {
  if (n <= 1) return n;
  return fib1(n - 1) + fib1(n - 2);
}

// Solución 2: Memoization (O(n))
function fib2(n, memo = {}) {
  if (n in memo) return memo[n];
  if (n <= 1) return n;
  
  memo[n] = fib2(n - 1, memo) + fib2(n - 2, memo);
  return memo[n];
}

// Solución 3: Iterative (O(n), mejor space complexity)
function fib3(n) {
  if (n <= 1) return n;
  
  let prev = 0;
  let curr = 1;
  
  for (let i = 2; i <= n; i++) {
    const next = prev + curr;
    prev = curr;
    curr = next;
  }
  
  return curr;
}
```

### SQL - QUERIES COMUNES

#### 1. Joins
```sql
-- Pregunta: Obtén usuarios con sus órdenes

-- INNER JOIN (solo usuarios con órdenes)
SELECT users.name, orders.total
FROM users
INNER JOIN orders ON users.id = orders.user_id;

-- LEFT JOIN (todos los usuarios, con o sin órdenes)
SELECT users.name, orders.total
FROM users
LEFT JOIN orders ON users.id = orders.user_id;

-- Usuarios SIN órdenes
SELECT users.name
FROM users
LEFT JOIN orders ON users.id = orders.user_id
WHERE orders.id IS NULL;
```

#### 2. Aggregations
```sql
-- Pregunta: Cuenta órdenes por usuario

SELECT 
  users.name,
  COUNT(orders.id) as order_count,
  SUM(orders.total) as total_spent,
  AVG(orders.total) as avg_order_value
FROM users
LEFT JOIN orders ON users.id = orders.user_id
GROUP BY users.id, users.name
HAVING COUNT(orders.id) > 0
ORDER BY total_spent DESC;
```

#### 3. Subqueries
```sql
-- Pregunta: Usuarios que gastaron más que el promedio

SELECT name, total_spent
FROM (
  SELECT 
    users.name,
    SUM(orders.total) as total_spent
  FROM users
  JOIN orders ON users.id = orders.user_id
  GROUP BY users.id, users.name
) as user_totals
WHERE total_spent > (
  SELECT AVG(total) FROM orders
);
```
